function getApiKey_() {
  return PropertiesService.getScriptProperties().getProperty("API_KEY") || "GIZLI_SIFRE_123";
}

// Sadece icinde "dd.mm" (gun.ay) gecen sayfalari tarar (Yil zorunlulugu kaldirildi).
// Boylece sayfa adi "17.08" de olsa, "17.08.2026" da olsa tum tebligat sayfalari taranir.
var TARIH_DESENI = new RegExp("\\d{2}\\.\\d{2}");
var YIL_DESENI = new RegExp("(?:^|\\.)(\\d{4})(?:$|\\.)");

// Arama, en yeniden eskiye doğru en fazla 16 tarih sayfasında yapılır.
// İlk eşleşmede durulmaz; bu pencere içindeki tüm eşleşmeler toplanır.
var ARAMA_SAYFA_LIMITI = 16;
var GET_ALL_CACHE_PREFIX = "tebligat_get_all_v1_";
var GET_ALL_CACHE_TTL_SECONDS = 300;
var GET_ALL_CACHE_CHUNK_SIZE = 45000;

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
      .setMimeType(ContentService.MimeType.JSON);
}

function hataDondur_(mesaj) {
  return json_({ success: false, error: mesaj });
}

function invalidateGetAllCache_() {
  var cache = CacheService.getScriptCache();
  var manifestKey = GET_ALL_CACHE_PREFIX + "manifest";
  var manifestText = cache.get(manifestKey);
  var keys = [manifestKey];

  if (manifestText) {
    try {
      var manifest = JSON.parse(manifestText);
      for (var i = 0; i < manifest.chunkCount; i++) {
        keys.push(GET_ALL_CACHE_PREFIX + i);
      }
    } catch (err) {
      // Bozuk manifest yalnızca manifest anahtarını temizlemeyi gerektirir.
    }
  }

  cache.removeAll(keys);
}

function readGetAllCache_() {
  var cache = CacheService.getScriptCache();
  var manifestText = cache.get(GET_ALL_CACHE_PREFIX + "manifest");
  if (!manifestText) return null;

  try {
    var manifest = JSON.parse(manifestText);
    var keys = [];
    for (var i = 0; i < manifest.chunkCount; i++) {
      keys.push(GET_ALL_CACHE_PREFIX + i);
    }

    var chunkMap = cache.getAll(keys);
    var allData = [];
    for (var c = 0; c < keys.length; c++) {
      if (!chunkMap[keys[c]]) return null;
      allData = allData.concat(JSON.parse(chunkMap[keys[c]]));
    }
    return allData;
  } catch (err) {
    invalidateGetAllCache_();
    return null;
  }
}

function writeGetAllCache_(allData) {
  var cache = CacheService.getScriptCache();
  var entries = {};
  var currentChunk = [];
  var currentSize = 2;
  var chunkCount = 0;

  var flushChunk = function() {
    if (currentChunk.length === 0) return;
    entries[GET_ALL_CACHE_PREFIX + chunkCount] = JSON.stringify(currentChunk);
    chunkCount++;
    currentChunk = [];
    currentSize = 2;
  };

  for (var i = 0; i < allData.length; i++) {
    var rowText = JSON.stringify(allData[i]);
    if (currentChunk.length > 0 && currentSize + rowText.length + 1 > GET_ALL_CACHE_CHUNK_SIZE) {
      flushChunk();
    }
    currentChunk.push(allData[i]);
    currentSize += rowText.length + 1;
  }
  flushChunk();

  entries[GET_ALL_CACHE_PREFIX + "manifest"] = JSON.stringify({ chunkCount: chunkCount });
  try {
    cache.putAll(entries, GET_ALL_CACHE_TTL_SECONDS);
  } catch (err) {
    // Cache başarısızlığı veri üretimini engellememeli.
  }
}

function parametreKontrol_(e, alanlar) {
  for (var i = 0; i < alanlar.length; i++) {
    var ad = alanlar[i];
    if (!e.parameter[ad] || e.parameter[ad].toString().trim() === "") {
      return ad;
    }
  }
  return null;
}

function doGet(e) {
  try {
    var apiKey = getApiKey_();
    if (!e || !e.parameter || e.parameter.key !== apiKey) {
      return hataDondur_("Yetkisiz erisim");
    }

    var action = e.parameter.action;
    var q = e.parameter.q;

    if (action === "getAll") return islemGetAll_(e);
    if (q || action === "search") return islemArama_(q || e.parameter.q || "", e.parameter.year);

    return hataDondur_("Gecersiz istek");
  } catch (err) {
    return hataDondur_("Sunucu hatasi: " + err.message);
  }
}

function doPost(e) {
  try {
    var payload = JSON.parse((e && e.postData && e.postData.contents) || "{}");
    if (!payload || typeof payload !== "object" || Array.isArray(payload)) {
      return hataDondur_("Gecersiz istek");
    }

    var apiKey = getApiKey_();
    if (!apiKey || payload.key !== apiKey) {
      return hataDondur_("Yetkisiz erisim");
    }

    var action = payload.action;
    var requestEvent = { parameter: payload };

    if (action === "unmark") {
      invalidateGetAllCache_();
      return islemUnmark_(requestEvent);
    }
    if (action === "add") {
      invalidateGetAllCache_();
      return islemAdd_(requestEvent);
    }
    if (action === "update") {
      invalidateGetAllCache_();
      return islemUpdate_(requestEvent);
    }
    if (action === "remove") {
      invalidateGetAllCache_();
      return islemRemove_(requestEvent);
    }
    if (action === "getAll") return islemGetAll_(requestEvent);
    if (action === "search") return islemArama_(payload.q || "", payload.year);

    return hataDondur_("Gecersiz istek");

  } catch (err) {
    console.error("Apps Script request failed.", err && err.name ? err.name : "UnknownError");
    return hataDondur_("Sunucu hatasi. Lutfen tekrar deneyin.");
  }
}

// ISARET KALDIRMA
// ------------------------------------------------------------
function islemUnmark_(e) {
  var eksik = parametreKontrol_(e, ["sayfa", "isim"]);
  if (eksik) return hataDondur_("Eksik parametre: " + eksik);

  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
  } catch (err) {
    return hataDondur_("Sistem mesgul, lutfen tekrar deneyin.");
  }

  try {
    var sayfa = (e.parameter.sayfa || "").trim();
    var isim = (e.parameter.isim || "").trim();
    var no = e.parameter.no !== undefined && e.parameter.no !== null ? String(e.parameter.no).trim() : "";

    var sheet = findSheetByNameOrDate_(SpreadsheetApp.getActiveSpreadsheet(), sayfa);
    if (!sheet) return hataDondur_("Sayfa bulunamadi: " + sayfa);

    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      var data = sheet.getRange(2, 1, lastRow - 1, 2).getValues();
      for (var i = 0; i < data.length; i++) {
        var rowIsim = String(data[i][0] || "").trim();
        var rowNo = String(data[i][1] || "").trim();
        var matchIsim = (rowIsim == isim);
        var matchNo = (!no || rowNo == no);
        if (matchIsim && matchNo) {
          var rowIndex = i + 2;
          sheet.getRange(rowIndex, 3).clearContent();
          sheet.getRange(rowIndex, 1).setBackground(null);
          return json_({ success: true, message: "Isaret kaldirildi" });
        }
      }
    }
    return hataDondur_("Kayit bulunamadi");

  } catch (err) {
    return hataDondur_("Isaret kaldirma hatasi: " + err.message);
  } finally {
    lock.releaseLock();
  }
}

// YARDIMCI: EN SON TARIHLI SAYFAYI BULMA
// ------------------------------------------------------------
function sonTarihliSayfayiBul_(ss) {
  var sheets = ss.getSheets();
  for (var i = sheets.length - 1; i >= 0; i--) {
    var name = sheets[i].getName();
    if (TARIH_DESENI.test(name)) {
      return sheets[i];
    }
  }
  return null;
}

// ------------------------------------------------------------
// YARDIMCI: SAYFAYI ISIM VEYA TARIH UYUMLULUGUYLA BULMA (dd.mm veya dd.mm.yyyy)
// ------------------------------------------------------------
function findSheetByNameOrDate_(ss, sayfaAdi) {
  if (!sayfaAdi) return null;
  var sheet = ss.getSheetByName(sayfaAdi);
  if (sheet) return sheet;

  // 1. "dd.mm.yyyy" formatindaysa "dd.mm" olarak dene
  var matchFull = sayfaAdi.match(/^(\d{2}\.\d{2})\.\d{4}$/);
  if (matchFull) {
    sheet = ss.getSheetByName(matchFull[1]);
    if (sheet) return sheet;
  }

  // 2. "dd.mm" formatindaysa aktif yilla "dd.mm.yyyy" olarak dene
  var matchShort = sayfaAdi.match(/^(\d{2}\.\d{2})$/);
  if (matchShort) {
    var yil = new Date().getFullYear();
    sheet = ss.getSheetByName(matchShort[1] + "." + yil);
    if (sheet) return sheet;
  }

  return null;
}

// ------------------------------------------------------------
// YARDIMCI: YENI TARIHLI SAYFA OLUSTURMA (SABLONDAN KOPYALAYIP TEMIZLEME)
// ------------------------------------------------------------
function yeniSayfaOlustur_(sayfaAdi) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var mevcut = ss.getSheetByName(sayfaAdi);
  if (mevcut) return mevcut;

  var sonSayfa = sonTarihliSayfayiBul_(ss);
  var yeniSheet;

  if (sonSayfa) {
    // Son sayfayi bicimlendirmesi, sutun genislikleri ve basliklariyla kopyala
    yeniSheet = sonSayfa.copyTo(ss);
    yeniSheet.setName(sayfaAdi);
    ss.setActiveSheet(yeniSheet);
    ss.moveActiveSheet(ss.getSheets().length);

    // 2. satirdan itibaren eski verileri ve renkleri temizle
    var lastRow = yeniSheet.getLastRow();
    var maxCols = Math.max(yeniSheet.getMaxColumns(), 3);
    if (lastRow > 1) {
      var veriAraligi = yeniSheet.getRange(2, 1, lastRow - 1, maxCols);
      veriAraligi.clearContent();
      veriAraligi.setBackground(null);
      veriAraligi.clearNote();
    }
  } else {
    // Tabloda hic tarihli sayfa yoksa sifirdan olustur
    yeniSheet = ss.insertSheet(sayfaAdi);
    yeniSheet.getRange(1, 1, 1, 2).setValues([["AD SOYAD", "NO"]]);
    yeniSheet.getRange(1, 1, 1, 2).setFontWeight("bold");
  }

  return yeniSheet;
}

// ------------------------------------------------------------
// OTOMATIK HAFTALIK SAYFA OLUSTURMA (PAZAR AKŞAMI TETIKLEYICISI ICIN)
// ------------------------------------------------------------
function otomatikHaftalikSayfaOlustur() {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
  } catch (err) {
    Logger.log("Kilit alinamadi: " + err.message);
    return { success: false, error: "Kilit alinamadi: " + err.message };
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sonSayfa = sonTarihliSayfayiBul_(ss);
    var yeniSayfaAdi = "";

    if (sonSayfa) {
      var sonAd = sonSayfa.getName().trim();
      var match = sonAd.match(/^(\d{2})\.(\d{2})(?:\.(\d{4}))?$/);
      if (match) {
        var gun = parseInt(match[1], 10);
        var ay = parseInt(match[2], 10) - 1;
        var yil = match[3] ? parseInt(match[3], 10) : new Date().getFullYear();

        var sonTarih = new Date(yil, ay, gun);
        // Son sayfadaki tarihe tam 7 gun (1 hafta) ekle
        sonTarih.setDate(sonTarih.getDate() + 7);

        var yeniGun = ("0" + sonTarih.getDate()).slice(-2);
        var yeniAy = ("0" + (sonTarih.getMonth() + 1)).slice(-2);
        var yeniYil = sonTarih.getFullYear();

        yeniSayfaAdi = match[3] ? (yeniGun + "." + yeniAy + "." + yeniYil) : (yeniGun + "." + yeniAy);
      }
    }

    // Eger son sayfa adindan tarih cikarilamazsa, sistem saatine gore sonraki Pazartesi'yi hesapla
    if (!yeniSayfaAdi) {
      var bugun = new Date();
      var gunIndex = bugun.getDay(); // 0: Pazar, 1: Pazartesi
      var eklenecekGun = (gunIndex === 0) ? 1 : (8 - gunIndex);
      var sonrakiPzt = new Date(bugun.getFullYear(), bugun.getMonth(), bugun.getDate() + eklenecekGun);

      var g = ("0" + sonrakiPzt.getDate()).slice(-2);
      var a = ("0" + (sonrakiPzt.getMonth() + 1)).slice(-2);
      var y = sonrakiPzt.getFullYear();
      yeniSayfaAdi = g + "." + a + "." + y;
    }

    // Eger sayfa zaten varsa tekrar olusturma
    if (ss.getSheetByName(yeniSayfaAdi)) {
      Logger.log("Sayfa zaten mevcut: " + yeniSayfaAdi);
      return { success: true, message: "Sayfa zaten mevcut", sayfa: yeniSayfaAdi };
    }

    yeniSayfaOlustur_(yeniSayfaAdi);
    invalidateGetAllCache_();
    Logger.log("Yeni haftalik sayfa basariyla olusturuldu: " + yeniSayfaAdi);
    return { success: true, message: "Sayfa olusturuldu", sayfa: yeniSayfaAdi };

  } catch (err) {
    Logger.log("Haftalik sayfa olusturma hatasi: " + err.message);
    return { success: false, error: err.message };
  } finally {
    lock.releaseLock();
  }
}

// ------------------------------------------------------------
// KAYIT EKLEME (AKILLI BOS SATIR BULMA VE NUMARA KORUMA)
// ------------------------------------------------------------
function islemAdd_(e) {
  var eksik = parametreKontrol_(e, ["sayfa", "isim"]);
  if (eksik) return hataDondur_("Eksik parametre: " + eksik);

  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
  } catch (err) {
    return hataDondur_("Sistem mesgul, lutfen tekrar deneyin.");
  }

  try {
    var sayfa = e.parameter.sayfa;
    var isim = e.parameter.isim;

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = findSheetByNameOrDate_(ss, sayfa);
    
    // Eger sayfa henuz acilmamissa otomatik olustur (hata vermesini onler)
    if (!sheet) {
      var sonSayfa = sonTarihliSayfayiBul_(ss);
      var hedefSayfaAdi = sayfa;
      if (sonSayfa && !YIL_DESENI.test(sonSayfa.getName())) {
        var kisaAd = sayfa.match(/^(\d{2}\.\d{2})/);
        if (kisaAd) hedefSayfaAdi = kisaAd[1];
      }
      sheet = yeniSayfaOlustur_(hedefSayfaAdi);
      if (!sheet) return hataDondur_("Bu haftanin sekmesi bulunamadi ve olusturulamadi.");
    }

    var lastRow = sheet.getLastRow();
    var targetRow = -1;
    var assignedNo = null;

    var data = null;
    // A sutununda (AD SOYAD) bos olan ilk satiri bul
    if (lastRow > 1) {
      data = sheet.getRange(2, 1, lastRow - 1, 2).getValues();
      for (var i = 0; i < data.length; i++) {
        var rowName = (data[i][0] !== null && data[i][0] !== undefined) ? data[i][0].toString().trim() : "";
        if (rowName === "") {
          targetRow = i + 2; // 1-tabanli satir numarasi
          var existingNo = data[i][1];
          if (existingNo !== "" && existingNo !== null && existingNo !== undefined) {
            assignedNo = existingNo;
          }
          break;
        }
      }
    }

    // Eger arada veya mevcut satirlar icinde bos isim bulunamadiysa, tablonun sonuna yeni satir ekle
    if (targetRow === -1) {
      targetRow = Math.max(lastRow + 1, 2);
    }

    // Eger B sutununda onceden suruklenmis/yazilmis bir numara yoksa, siradaki numarayi hesapla
    if (assignedNo === null || assignedNo === "") {
      var maxNo = 0;
      if (lastRow > 1 && data) {
        for (var k = 0; k < data.length; k++) {
          var num = parseInt(data[k][1], 10);
          if (!isNaN(num) && num > maxNo) maxNo = num;
        }
      }
      assignedNo = maxNo > 0 ? (maxNo + 1) : (targetRow - 1);

      // Hem ismi hem yeni numarayi yaz
      sheet.getRange(targetRow, 1, 1, 2).setValues([[isim, assignedNo]]);
    } else {
      // B sutununda numara zaten vardi, sadece A sutunundaki hucreye ismi yaz (numaraya dokunma)
      sheet.getRange(targetRow, 1).setValue(isim);
    }

    // Satirin isim hucresindeki arka plan rengini temizle (eski sablon rengi kalmissa)
    sheet.getRange(targetRow, 1).setBackground(null);

    return json_({ success: true, message: "Kayit eklendi", assignedNo: assignedNo, sayfa: sheet.getName() });

  } catch (err) {
    return hataDondur_("Kayit ekleme hatasi: " + err.message);
  } finally {
    lock.releaseLock();
  }
}

// ------------------------------------------------------------
// ARAMA
// ------------------------------------------------------------
function islemArama_(q, year) {
  try {
    var rawQ = q.toString();
    q = rawQ.toLowerCase();
    year = /^\d{4}$/.test(year || "") ? year : String(new Date().getFullYear());
    var sonuclar = [];
    var benzersizKontrol = {};

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    // Yil filtresi kaynakta uygulanir; yili yazili 2025/2024 sekmeleri okunmaz.
    // Yilsiz "dd.mm" sekmeler mevcut yil olarak kabul edilir.
    var tarihliSayfalar = ss.getSheets().filter(function(sheet) {
      var sheetName = sheet.getName();
      var yearMatch = sheetName.match(YIL_DESENI);
      return TARIH_DESENI.test(sheetName) && (!yearMatch || yearMatch[1] === year);
    }).reverse();

    // Tum dosyadaki TextFinder taramasi buyuk tablolarda web uygulamasini zaman asimina sokuyordu.
    // En yeni 16 sayfada yalnizca gerekli uc sutun okunur ve pencere icindeki
    // tum eslesmeler toplanir.
    var taranacakSayfaSayisi = Math.min(tarihliSayfalar.length, ARAMA_SAYFA_LIMITI);
    for (var s = 0; s < taranacakSayfaSayisi; s++) {
      var sheet = tarihliSayfalar[s];
      var sayfaAdi = sheet.getName();
      
      var lastRow = sheet.getLastRow();
      if (lastRow <= 1) continue;
      
      var data = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
      
      for (var r = 0; r < data.length; r++) {
        var rowData = data[r];
        var isim = rowData[0] ? rowData[0].toString().toLowerCase() : "";
        
        if (isim.indexOf(q) !== -1) {
          var isaretliMi = (rowData[2] !== "" && rowData[2] !== null && rowData[2] !== undefined);
          var no = rowData[1];
          var anahtar = sayfaAdi + "_" + no;
          
          if (!benzersizKontrol[anahtar]) {
            benzersizKontrol[anahtar] = true;
            sonuclar.push({
              sayfa: sayfaAdi,
              isim: rowData[0],
              no: no,
              isaretli: isaretliMi
            });
          }
        }
      }

    }
    
    return json_({ results: sonuclar });

  } catch (err) {
    return hataDondur_("Arama hatasi: " + err.message);
  }
}

// ------------------------------------------------------------
// TUM VERIYI BULUTA CEKME (CACHE ICIN)
// ------------------------------------------------------------
function islemGetAll_(e) {
  try {
    var cachedData = readGetAllCache_();
    if (cachedData !== null) {
      return json_({ success: true, results: cachedData, cached: true });
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheets = ss.getSheets();
    var allData = [];
    
    // Tum sayfalari gez
    for (var s = 0; s < sheets.length; s++) {
      var sheet = sheets[s];
      var sheetName = sheet.getName();
      
      // Sadece tarihli sayfalari al
      if (!TARIH_DESENI.test(sheetName)) continue;
      
      var lastRow = sheet.getLastRow();
      if (lastRow <= 1) continue;
      
      // Her sayfadaki datalari oku (A, B, C)
      var data = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
      for (var r = 0; r < data.length; r++) {
        var isim = data[r][0];
        if (isim) {
          allData.push({
            sayfa: sheetName,
            isim: isim,
            no: data[r][1],
            isaretli: (data[r][2] !== "" && data[r][2] !== null && data[r][2] !== undefined)
          });
        }
      }
    }
    
    writeGetAllCache_(allData);
    return json_({ success: true, results: allData, cached: false });
  } catch (err) {
    return hataDondur_("GetAll hatasi: " + err.message);
  }
}
